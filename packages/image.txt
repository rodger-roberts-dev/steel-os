# Steel OS: rpm packages added to the system image (read by build_files/build.sh)
# Fedora package names. One per line. Only system pieces go here; apps are
# Flatpaks (see flatpaks-default.txt and steel-apps.txt).


# Printing for older, non-driverless printers (STEEL-3 findings).
hplip
gutenprint-cups

# Makes this machine appear in Windows Explorer's Network, and helps Dolphin
# find Windows PCs (STEEL-3 findings).
wsdd

# Verify after the first boot whether Bazzite already includes these KDE apps.
# Add any that are missing:
#   ark kate okular skanpage spectacle gwenview kolourpaint print-manager

# Virtual machines out of the box: the libvirt service, QEMU/KVM, the default
# NAT network and a software TPM (for Windows 11 guests). Bazzite does not
# ship these. Virtual Machine Manager itself is a Steel Apps one-click Flatpak.
libvirt-daemon-kvm
libvirt-daemon-config-network
swtpm

# First-login setup dialogs (STEEL-35). Usually already part of KDE.
kdialog

# Password reset window before the login screen (STEEL-36): a minimal
# full-screen session, and text-mode dialogs as a fallback.
cage
dialog

# Login screen (STEEL-36): SDDM with the Breeze theme, so Steel OS can add
# "Forgot your password?". Replaces Fedora 44's Plasma Login Manager.
sddm
sddm-breeze
sddm-wayland-plasma

# Printers and scanners, the common brands without hunting for drivers (STEEL-38).
# HP: hplip (above). Canon and Epson: gutenprint-cups (above) and driverless.
# Brother lasers: brlaser. Most printers since about 2015: driverless (IPP
# Everywhere / AirPrint), over the network or USB (ipp-usb).
printer-driver-brlaser
ipp-usb
cups-browsed
foomatic
foomatic-db-ppds
splix
# Plug in a USB printer and it is set up by itself.
system-config-printer-udev
# Scanning: Skanpage app, SANE drivers (Canon PIXMA, Epson, many more),
# Epson scanner firmware, and driverless network scanners (eSCL / WSD, used by
# most Brother, Canon, Epson and HP models with Wi-Fi).
skanpage
sane-backends-drivers-scanners
sane-airscan
iscan-firmware

# Windows networking (STEEL-39): be seen by Windows PCs (wsdd, above), open
# their shares in Dolphin, and share a folder with a right-click (Samba
# usershares + Dolphin's Share tab).
samba
samba-usershares
kdenetwork-filesharing
