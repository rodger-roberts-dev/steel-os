# Steel OS approved optional applications
# Not installed by default. Offered one click at a time from the Steel Apps
# install menu. Every line is a real package name. Sources and install
# commands are in docs/app-install-plan.md.

# Not packages, so not listed here (see docs/app-install-plan.md section 2):
#   Battle.net            omarchy install gaming battlenet
#   Google web apps       omarchy webapp install <name> <url>

# Omarchy Package Repository (pkgs.omarchy.org), not the AUR.
# Steel OS default browser. Install with: omarchy install browser chrome
google-chrome

# Arch extra. One-click install required (T, 2026-10-15); also rodger-roberts-dev's
# browser. Install with: omarchy install browser firefox
firefox

# Arch extra. Already part of the Omarchy base install.
obsidian

# Arch extra.
discord

# AUR only. Prefer the Zoom web app (Omarchy has a built-in handler for it).
# Keep this as the fallback for people who need the desktop client.
zoom

# Omarchy Package Repository. One-click install required (T, 2026-10-15).
# Install with: omarchy install app Slack slack-desktop
slack-desktop

# Omarchy Package Repository. One-click install required (T, 2026-10-15).
# Install with: omarchy install service dropbox (also pulls dropbox-cli and
# nautilus-dropbox). Its Omarchy top-bar plugin does not apply on Plasma; the
# Dropbox tray icon works there instead.
dropbox

# Arch extra. Already part of the Omarchy base install. PC game streaming client.
moonlight-qt

# AUR. One-click install required (T, 2026-10-15). Scanner software.
# Install with: omarchy pkg aur add naps2-bin   (Skanpage in core is the built-in alternative)
naps2-bin

# AUR. One-click install (T, 2026-10-15). A Linux remake of Notepad++.
# Install with: omarchy pkg aur add notepadnext-bin   (Kate in core is the built-in editor)
notepadnext-bin

# Arch extra. Runs the EA app (formerly Origin), Ubisoft Connect and other
# Windows game stores through Wine. Install with: omarchy install gaming lutris
lutris
